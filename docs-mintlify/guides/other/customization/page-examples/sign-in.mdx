Examples for custom sign-in pages using Hexclave components and SDK methods. Configure your custom URL in [Custom Pages](/guides/other/customization/custom-pages).

## Custom page with `SignIn` component

```tsx title="app/signin/page.tsx"
"use client";

import { SignIn } from "@hexclave/next";

export default function DefaultSignIn() {
  // Optionally redirect if the user is already signed in:
  // const user = useUser();
  // if (user) { redirect("/dashboard"); }
  return <SignIn fullPage />;
}
```

You can call `useUser()` at the top of the page to detect an existing session and redirect before rendering the form.

## Other useful components

- **`CredentialSignIn`**: Email-and-password sign-in form with validation and error handling.
- **`OAuthButtonGroup`**: Renders OAuth provider buttons based on your project configuration.
- **`OAuthButton`**: A single OAuth provider button when you want to offer only specific providers.

## Custom OAuth sign-in

```tsx title="app/signin/page.tsx"
"use client";

import { useHexclaveApp } from "@hexclave/next";

export default function CustomOAuthSignIn() {
  const app = useHexclaveApp();

  return (
    <div>
      <h1>My custom sign-in page</h1>
      <button
        type="button"
        onClick={async () => {
          await app.signInWithOAuth("google");
        }}
      >
        Sign in with Google
      </button>
    </div>
  );
}
```

## Custom credential sign-in

```tsx title="app/signin/page.tsx"
"use client";

import { useHexclaveApp } from "@hexclave/next";
import { useState } from "react";

export default function CustomCredentialSignIn() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const app = useHexclaveApp();

  const onSubmit = async () => {
    if (!password) {
      setError("Please enter your password");
      return;
    }

    const result = await app.signInWithCredential({ email, password });
    if (result.status === "error") {
      setError(result.error.humanReadableMessage);
    }
    // On success, Hexclave redirects to urls.afterSignIn (configurable in HexclaveServerApp).
  };

  return (
    <form onSubmit={(e) => { e.preventDefault(); onSubmit(); }}>
      {error && <div>{error}</div>}
      <input
        type="email"
        placeholder="email@example.com"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
      />
      <input
        type="password"
        placeholder="password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
      />
      <button type="submit">Sign in</button>
    </form>
  );
}
```

Prefer checking `result.error.errorCode` for specific error UI — see [Known Errors](/guides/other/known-errors).

## Custom magic link sign-in

```tsx title="app/signin/page.tsx"
"use client";

import { useHexclaveApp } from "@hexclave/next";
import { useState } from "react";

export default function CustomMagicLinkSignIn() {
  const [email, setEmail] = useState("");
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");
  const app = useHexclaveApp();

  const onSubmit = async () => {
    if (!email) {
      setError("Please enter your email address");
      return;
    }

    const result = await app.sendMagicLinkEmail(email);
    if (result.status === "error") {
      setError(result.error.humanReadableMessage);
    } else {
      setMessage("Magic link sent! Please check your email.");
    }
  };

  return (
    <form onSubmit={(e) => { e.preventDefault(); onSubmit(); }}>
      {error && <div>{error}</div>}
      {message ? (
        <div>{message}</div>
      ) : (
        <>
          <input
            type="email"
            placeholder="email@example.com"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
          />
          <button type="submit">Send magic link</button>
        </>
      )}
    </form>
  );
}
```
